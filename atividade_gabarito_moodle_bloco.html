1Triagem automatica de tickets com operador de resto
programa
{
	funcao inicio()
	{
		inteiro numero

		escreva("Digite um numero inteiro: ")
		leia(numero)

		se (numero % 2 == 0)
		{
			escreva("O numero e par")
		}
		senao
		{
			escreva("O numero e impar")
		}
	}
}
2Regra comercial de desconto no fechamento da venda
programa
{
	funcao inicio()
	{
		real valor_compra, valor_final

		escreva("Digite o valor da compra: ")
		leia(valor_compra)

		se (valor_compra >= 500)
		{
			valor_final = valor_compra * 0.90
			escreva("Desconto aplicado. Valor final: ", valor_final)
		}
		senao
		{
			valor_final = valor_compra
			escreva("Sem desconto. Valor final: ", valor_final)
		}
	}
}

5Modulo de calculo rapido para atendimento comercial
programa
{
	funcao inicio()
	{
		inteiro opcao
		real a, b, resultado

		escreva("=== Calculadora Simples ===")
		escreva("1 - Somar")
		escreva("2 - Subtrair")
		escreva("3 - Multiplicar")
		escreva("4 - Dividir")
		escreva("Escolha a opcao: ")
		leia(opcao)

		escreva("Digite o primeiro numero: ")
		leia(a)
		escreva("Digite o segundo numero: ")
		leia(b)

		escolha (opcao)
		{
			caso 1:
				resultado = a + b
				escreva("Resultado: ", resultado)
			pare
			caso 2:
				resultado = a - b
				escreva("Resultado: ", resultado)
			pare
			caso 3:
				resultado = a * b
				escreva("Resultado: ", resultado)
			pare
			caso 4:
				se (b != 0)
				{
					resultado = a / b
					escreva("Resultado: ", resultado)
				}
				senao
				{
					escreva("Nao e possivel dividir por zero")
				}
			pare
			caso contrario:
				escreva("Opcao invalida")
		}
	}
}
6Classificacao de desempenho para relatorio gerencial
programa
{
	funcao inicio()
	{
		inteiro nota
		cadeia conceito

		escreva("Digite a nota de 0 a 10: ")
		leia(nota)

		se (nota < 0 ou nota > 10)
		{
			escreva("Nota invalida")
		}
		senao
		{
			escolha (nota)
			{
				caso 10:
				caso 9:
					conceito = "A"
				pare
				caso 8:
				caso 7:
					conceito = "B"
				pare
				caso 6:
					conceito = "C"
				pare
				caso 5:
					conceito = "D"
				pare
				caso contrario:
					conceito = "E"
			}

			escreva("Conceito final: ", conceito)
		}
	}
}
7Geracao sequencial para senhas de atendimento
programa
{
	funcao inicio()
	{
		inteiro limite, contador

		escreva("Contar ate qual numero? ")
		leia(limite)

		contador = 1

		enquanto (contador <= limite)
		{
			escreva(contador, " ")
			contador = contador + 1
		}
	}
}
