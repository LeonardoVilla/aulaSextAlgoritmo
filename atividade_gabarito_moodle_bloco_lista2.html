8Monitoramento manual de temperatura em servidores
programa
{
	funcao inicio()
	{
		inteiro contador
		real temperatura

		contador = 1

		enquanto (contador <= 5)
		{
			escreva("Digite a temperatura do servidor ", contador, ": ")
			leia(temperatura)

			se (temperatura > 70)
			{
				escreva("ALERTA: temperatura acima do limite")
			}
			senao
			{
				escreva("OK: temperatura em faixa segura")
			}

			contador = contador + 1
		}
	}
}
9Impressao sequencial de etiquetas por lote
programa
{
	funcao inicio()
	{
		inteiro total, etiqueta

		escreva("Quantidade de etiquetas do lote: ")
		leia(total)

		para (etiqueta = 1; etiqueta <= total; etiqueta = etiqueta + 1)
		{
			escreva("Etiqueta #", etiqueta)
		}
	}
}
10Contagem de pares e impares em intervalo numerico
programa
{
	funcao inicio()
	{
		inteiro inicio, fim, numero, pares, impares

		escreva("Digite o inicio do intervalo: ")
		leia(inicio)
		escreva("Digite o fim do intervalo: ")
		leia(fim)

		pares = 0
		impares = 0

		para (numero = inicio; numero <= fim; numero = numero + 1)
		{
			se (numero % 2 == 0)
			{
				pares = pares + 1
			}
			senao
			{
				impares = impares + 1
			}
		}

		escreva("Quantidade de pares: ", pares)
		escreva("Quantidade de impares: ", impares)
	}
}
11Controle de tentativas de login em terminal local
programa
{
	funcao inicio()
	{
		inteiro tentativas
		cadeia senha
		logico autenticado

		tentativas = 0
		autenticado = falso

		enquanto (tentativas < 3 e autenticado == falso)
		{
			escreva("Digite a senha: ")
			leia(senha)

			se (senha == "1234")
			{
				autenticado = verdadeiro
				escreva("Acesso liberado")
			}
			senao
			{
				tentativas = tentativas + 1
				escreva("Senha incorreta. Tentativas usadas: ", tentativas)
			}
		}

		se (autenticado == falso)
		{
			escreva("Acesso bloqueado")
		}
	}
}
12Calculo de media final com decisao de aprovacao
programa
{
	funcao inicio()
	{
		inteiro i
		real nota, soma, media

		soma = 0

		para (i = 1; i <= 4; i = i + 1)
		{
			escreva("Digite a nota ", i, ": ")
			leia(nota)
			soma = soma + nota
		}

		media = soma / 4
		escreva("Media final: ", media)

		se (media >= 7)
		{
			escreva("Situacao: aprovado")
		}
		senao
		{
			escreva("Situacao: reprovado")
		}
	}
}
